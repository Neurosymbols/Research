#mireot -> gets the required properties/classes with their hierarchy. Does not bring characteristics and relations.
robot extract --method MIREOT --input ./app/data/ontologies/epoch3-7/ro.owl --upper-terms ./app/data/input/epoch3-7/mireot-ro-upper-terms.txt --lower-terms ./app/data/input/epoch3-7/mireot-ro-lower-terms.txt --output ./app/data/ontologies/epoch3-7/ro-causal-properties-hie.owl --imports exclude -vvv

#subset -> method used because gets the required properties/classes with their characteristics and relations between them. Does not bring hierarchy
robot extract --method SUBSET --input ./app/data/ontologies/epoch3-7/ro.owl  --term-file ./app/data/input/epoch3-7/subset-ro-terms.txt --output ./app/data/ontologies/epoch3-7/ro-causal-properties-logical.owl --imports exclude -vvv

robot merge \
  --input ./app/data/ontologies/epoch3-7/ro-causal-properties-hie.owl \
  --input ./app/data/ontologies/epoch3-7/ro-causal-properties-logical.owl \
  --output ./app/data/ontologies/epoch3-7/ro-causal-properties.owl

wget https://github.com/ontodev/robot/releases/download/v1.9.7/robot.jar
